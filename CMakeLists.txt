cmake_minimum_required(VERSION 3.24)
project(Veyra VERSION 0.9.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required.")
endif()
add_executable(veyra WIN32 src/main.cpp app.rc)
target_include_directories(veyra PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(veyra PRIVATE cxx_std_20)
target_compile_definitions(veyra PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(veyra PRIVATE user32 gdi32 bcrypt dwmapi gdiplus)
if(MSVC)
  target_compile_options(veyra PRIVATE /W4 /WX /permissive- /utf-8)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core-tests tests/core_tests.cpp)
  target_compile_features(core-tests PRIVATE cxx_std_20)
  add_test(NAME core COMMAND core-tests)
  add_executable(local-pass-fixture tests/local_pass_fixture.cpp)
  target_compile_features(local-pass-fixture PRIVATE cxx_std_20)
  target_compile_definitions(local-pass-fixture PRIVATE FOUNDATION_PATCH_TESTS UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(local-pass-fixture PRIVATE bcrypt)
  add_test(NAME local-pass-memory COMMAND local-pass-fixture)
endif()
